git clone https://github.com/borjatarraso/gnu-emacs-os.git
cd gnu-emacs-os
make -C pid1 all
make -C shstub
set QCOW (guix time-machine -C guix-system/channels.scm -- system image -t qcow2 -L $PWD guix-system/system.scm | tail -n 1)
echo $QCOW
qemu-system-x86_64 -enable-kvm -m 2048 -cpu host -smp 2 -vga virtio -display gtk -device qemu-xhci,id=xhci -device usb-tablet,bus=xhci.0 -serial mon:stdio -snapshot -drive file=$QCOW,format=qcow2,if=virtio
